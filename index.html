function parseVisualGridCSV(text) { const lines = text.split('\n').map(l => l.trim().replace(/^"|"$/g, '')); // Define column indexes where your team names appear in row 9 (0-indexed) // C=2, I=8, O=14, U=20 based on typical sheet layout const teamColumns = [ { nameIdx: 2, playerIdx: 3, roleIdx: 2 }, // Empress of Light column { nameIdx: 8, playerIdx: 9, roleIdx: 8 }, // January column { nameIdx: 14, playerIdx: 15, roleIdx: 14 },// Reaper column { nameIdx: 20, playerIdx: 21, roleIdx: 20 } // Wingman column ]; rostersData = {}; dynamicTeams = []; // Row 9 contains team names (e.g. Empress of Light | 1034) if (lines.length > 8) { const headerRow = lines[8].split(','); teamColumns.forEach(col => { let rawName = headerRow[col.nameIdx]; if (rawName && rawName.trim()) { let cleanName = rawName.split('|')[0].trim(); // Removes the "| 1034" dynamicTeams.push(cleanName); rostersData[cleanName] = []; col.cleanName = cleanName; } }); } // Read player rows starting from row 10 downwards lines.slice(9).forEach(line => { const parts = line.split(','); teamColumns.forEach(col => { if (!col.cleanName) return; let player = parts[col.playerIdx] ? parts[col.playerIdx].trim() : ''; let role = parts[col.roleIdx] ? parts[col.roleIdx].trim() : 'Player'; // Filter out empty lines or rank numbers if (player && player !== '' && isNaN(player)) { rostersData[col.cleanName].push({ name: player, role: role }); } }); }); // Register teams into standings dynamicTeams.forEach(name => { if (!teamsData.find(t => t.name === name)) { teamsData.push({ name: name, wins: 0, losses: 0 }); } }); saveData(); populateDropdowns(); renderStandings(); renderRosters(); }
